def now():
    return int(time.time())


def get_user(user_id):
    cursor.execute(
        "SELECT * FROM users WHERE user_id = ?",
        (user_id,)
    )
    return cursor.fetchone()


def ensure_user(user_id):
    if not get_user(user_id):
        cursor.execute("""
        INSERT INTO users (user_id, created_at)
        VALUES (?, ?)
        """, (user_id, now()))
        db.commit()


def is_banned(user_id):
    user = get_user(user_id)
    return bool(user and user[9])


def premium_active(user_id):
    user = get_user(user_id)
    return bool(user and user[5] > now())


def priority_active(user_id):
    user = get_user(user_id)
    return bool(user and user[6] > now())


def interests_active(user_id):
    user = get_user(user_id)
    return bool(user and user[7] > now())


def themes_active(user_id):
    user = get_user(user_id)
    return bool(user and user[8] > now())


def get_gender(user_id):
    user = get_user(user_id)
    return user[1] if user else "unknown"


def get_search_gender(user_id):
    user = get_user(user_id)
    return user[2] if user else "any"


def get_interests(user_id):
    user = get_user(user_id)
    if not user or not user[3]:
        return []

    return user[3].split(",")


def remove_from_queue(user_id):
    try:
        waiting.remove(user_id)
    except ValueError:
        pass


def blocked(user1, user2):
    cursor.execute("""
    SELECT 1 FROM blocks
    WHERE user_id = ? AND blocked_id = ?
    """, (user1, user2))

    return cursor.fetchone() is not None


def compatible(user1, user2):

    if user1 == user2:
        return False

    if blocked(user1, user2) or blocked(user2, user1):
        return False

    if is_banned(user1) or is_banned(user2):
        return False

    gender1 = get_gender(user1)
    gender2 = get_gender(user2)

    wanted1 = get_search_gender(user1)
    wanted2 = get_search_gender(user2)

    # Первый пользователь выбирает пол
    if wanted1 != "any":
        if wanted1 != gender2:
            return False

    # Второй пользователь тоже выбирает пол
    if wanted2 != "any":
        if wanted2 != gender1:
            return False

    return True

@dp.callback_query(F.data.startswith("toggle_interest_"))
async def toggle_interest(callback: CallbackQuery):

    user_id = callback.from_user.id

    if not interests_active(user_id):
        await callback.answer(
            "Сначала купи функцию интересов.",
            show_alert=True
        )
        return

    key = callback.data.replace(
        "toggle_interest_",
        ""
    )

    current = get_interests(user_id)

    if key in current:
        current.remove(key)
    else:
        if len(current) >= 5:
            await callback.answer(
                "Можно выбрать максимум 5 интересов.",
                show_alert=True
            )
            return

        current.append(key)

    cursor.execute(
        "UPDATE users SET interests = ? WHERE user_id = ?",
        (",".join(current), user_id)
    )

    db.commit()

    await callback.answer("Обновлено ✅")

@dp.callback_query(F.data == "search")
async def search(callback: CallbackQuery):

    user_id = callback.from_user.id

    if is_banned(user_id):
        await callback.answer(
            "Доступ заблокирован.",
            show_alert=True
        )
        return

    if user_id in pairs:
        await callback.answer(
            "У тебя уже есть собеседник.",
            show_alert=True
        )
        return

    remove_from_queue(user_id)

    # Premium priority
    if priority_active(user_id) or premium_active(user_id):
        waiting.appendleft(user_id)
    else:
        waiting.append(user_id)

    await callback.message.edit_text(
        "🔎 <b>ИЩУ СОБЕСЕДНИКА...</b>\n\n"
        "Подожди немного 👀",
        parse_mode="HTML",
        reply_markup=InlineKeyboardMarkup(
            inline_keyboard=[
                [
                    InlineKeyboardButton(
                        text="❌ Отменить поиск",
                        callback_data="cancel_search"
                    )
                ]
            ]
        )
    )

    await find_matches()

    await callback.answer()


async def find_matches():

    changed = True

    while changed:

        changed = False

        if len(waiting) < 2:
            return

        users = list(waiting)

        for user1 in users:

            if user1 not in waiting:
                continue

            for user2 in users:

                if user1 == user2:
                    continue

                if user2 not in waiting:
                    continue

                if compatible(user1, user2):

                    remove_from_queue(user1)
                    remove_from_queue(user2)

                    pairs[user1] = user2
                    pairs[user2] = user1

                    await bot.send_message(
                        user1,
                        "💚 <b>СОБЕСЕДНИК НАЙДЕН!</b>\n\n"
                        "Можете начинать общение.",
                        parse_mode="HTML",
                        reply_markup=chat_menu(),
                        protect_content=True
                    )

                    await bot.send_message(
                        user2,
                        "💚 <b>СОБЕСЕДНИК НАЙДЕН!</b>\n\n"
                        "Можете начинать общение.",
                        parse_mode="HTML",
                        reply_markup=chat_menu(),
                        protect_content=True
                    )

                    changed = True
                    break

            if changed:
                break

@dp.pre_checkout_query()
async def pre_checkout(query):

    payload = query.invoice_payload

    allowed = {
        "premium_30",
        "priority_30",
        "interests_30",
        "themes_30",
        "reactions_10"
    }

    if payload not in allowed:

        await query.answer(
            ok=False,
            error_message="Неизвестный товар."
        )

        return

    await query.answer(ok=True)

@dp.message(F.successful_payment)
async def successful_payment(message: Message):

    user_id = message.from_user.id
    payment = message.successful_payment

    payload = payment.invoice_payload
    amount = payment.total_amount
    charge_id = payment.telegram_payment_charge_id

    # Сохраняем оплату
    cursor.execute("""
    INSERT INTO payments
    (user_id, payload, amount, charge_id, created_at)
    VALUES (?, ?, ?, ?, ?)
    """, (
        user_id,
        payload,
        amount,
        charge_id,
        now()
    ))

    # Важно: успешная оплата подтверждается
    # именно этим событием Telegram.
    if payload == "premium_30":

        old = get_user(user_id)[5]
        start = max(old, now())
        until = start + MONTH

        cursor.execute("""
        UPDATE users
        SET premium_until = ?,
            priority_until = ?,
            interests_until = ?,
            themes_until = ?
        WHERE user_id = ?
        """, (
            until,
            until,
            until,
            until,
            user_id
        ))

        await message.answer(
            "🎉 <b>Premium активирован!</b>\n\n"
            "⭐ Теперь тебе доступны все Premium-функции "
            "на 30 дней.",
            parse_mode="HTML",
            reply_markup=main_menu(),
            protect_content=True
        )

    elif payload == "priority_30":

        old = get_user(user_id)[6]
        start = max(old, now())

        cursor.execute(
            "UPDATE users SET priority_until = ? WHERE user_id = ?",
            (start + MONTH, user_id)
        )

        await message.answer(
            "🔥 Приоритетный поиск активирован на 30 дней!",
            reply_markup=main_menu(),
            protect_content=True
        )

    elif payload == "interests_30":

        old = get_user(user_id)[7]

        cursor.execute(

        await message.answer(
            "🎯 Интересы разблокированы на 30 дней!",

        )

    elif payload == "themes_30":

        old = get_user(user_id)[8]
        start = max(old, now())

        cursor.execute(
            "UPDATE users SET themes_until = ? WHERE user_id = ?",
            (start + MONTH, user_id)
        )

        await message.answer(
            "🎨 Premium-темы разблокированы на 30 дней!",
            reply_markup=main_menu(),

        cursor.execute(

@dp.message()
async def messages(message: Message):

    user_id = message.from_user.id

    ensure_user(user_id)

    if is_banned(user_id):
        return

    partner = pairs.get(user_id)

    if not partner:

        await message.answer(
            "🔎 Сейчас у тебя нет собеседника.\n\n"
            "Нажми «Найти собеседника».",
            reply_markup=main_menu(),
            protect_content=True
        )

        return

    # Только передаём сообщение через бота.
    try:

        await bot.copy_message(
            chat_id=partner,
            from_chat_id=message.chat.id,
            message_id=message.message_id,
            protect_content=True
        )

    except TelegramBadRequest:

        await message.answer(
            "⚠️ Не удалось отправить это сообщение.",
            protect_content=True
        )
